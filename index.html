```javascript const { onValueCreated } = require("firebase-functions/v2/database"); const { initializeApp } = require("firebase-admin/app"); const { getDatabase } = require("firebase-admin/database"); initializeApp(); /* ==================================================== PUNTI AUTOMATICI ==================================================== */ const PUNTI = { quiz: 10, pronostico: 10, foto_challenge: 15, sfida_invitati: 10 }; /* ==================================================== QUANDO VIENE REGISTRATO UN GIOCO ==================================================== */ exports.assegnaPuntiGioco = onValueCreated( "/giochi/{giocoId}", async (event) => { const gioco = event.data.val(); const giocoId = event.params.giocoId; if (!gioco) { return; } const squadra = gioco.squadra; const tipo = gioco.tipo; if (!squadra || !tipo) { console.log( "Gioco non valido:", giocoId ); return; } const punti = PUNTI[tipo]; if ( typeof punti !== "number" ) { console.log( "Tipo di gioco sconosciuto:", tipo ); return; } const db = getDatabase(); /* ================================================ PROTEZIONE CONTRO DOPPIA ELABORAZIONE ================================================ */ const elaboratoRef = db.ref( "giochiElaborati/" + giocoId ); const risultato = await elaboratoRef.transaction( function(valore){ if(valore === true){ return; } return true; } ); /* Se era già stato elaborato, non assegniamo nuovamente i punti. */ if(!risultato.committed){ console.log( "Gioco già elaborato:", giocoId ); return; } /* ================================================ AGGIUNGI I PUNTI ALLA SQUADRA ================================================ */ const squadraRef = db.ref( "squadre/" + squadra ); await squadraRef.transaction( function(s){ if(!s){ return s; } s.punti = Number(s.punti || 0) + punti; return s; } ); /* ================================================ REGISTRAZIONE AZIONE ================================================ */ await db.ref("azioni").push({ squadra:squadra, punti:punti, tipo:tipo, giocoId:giocoId, giocatore: gioco.giocatore || "Invitato", automatica:true, data: Date.now() }); console.log( "Assegnati " + punti + " punti alla squadra " + squadra ); } ); /* ==================================================== MODIFICHE MANUALI ORGANIZZATORE ==================================================== */ exports.modificaPunteggioAdmin = onValueCreated( "/adminActions/{actionId}", async (event) => { const action = event.data.val(); if(!action){ return; } const email = action.email; /* ================================================== IMPORTANTE: SOSTITUISCI QUESTA EMAIL CON LA TUA EMAIL ORGANIZZATRICE ================================================== */ const EMAIL_ADMIN = "INSERISCI-QUI-LA-TUA-EMAIL"; if( !email || email !== EMAIL_ADMIN ){ console.log( "Modifica rifiutata:", email ); return; } const squadra = action.squadra; const punti = Number(action.punti); if( !squadra || !Number.isFinite(punti) ){ return; } const db = getDatabase(); await db.ref( "squadre/" + squadra ).transaction( function(s){ if(!s){ return s; } s.punti = Number(s.punti || 0) + punti; return s; } ); await db.ref("azioni").push({ squadra:squadra, punti:punti, descrizione: action.descrizione || "Modifica organizzatore", organizzatore:email, manuale:true, data:Date.now() }); console.log( "Modifica admin:", squadra, punti ); } ); ```
